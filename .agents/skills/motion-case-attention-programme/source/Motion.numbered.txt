0112 // 本轮语义点共用切线连接，源场景时刻不因镜片路径改变。
0113 function ReadingPoint(p:any,f:number){
0114  const nodes=p.lensNodes;
0115  if(f<=nodes[0].f)return [nodes[0].x,nodes[0].y];
0116  const last=nodes.length-1;
0117  if(f>=nodes[last].f)return [nodes[last].x,nodes[last].y];
0118  let k=0;while(k<last-1&&f>=nodes[k+1].f)k++;
0119  const duration=nodes[k+1].f-nodes[k].f,t=(f-nodes[k].f)/duration;
0120  const tangent=(i:number,key:string)=>{
0121  if(i===last)return 0;
0122  if(i===0)return (nodes[1][key]-nodes[0][key])/(nodes[1].f-nodes[0].f)*.7;
0123  return (nodes[i+1][key]-nodes[i-1][key])/(nodes[i+1].f-nodes[i-1].f);
0124  };
0125  return ['x','y'].map(key=>(2*t*t*t-3*t*t+1)*nodes[k][key]+(t*t*t-2*t*t+t)*duration*tangent(k,key)+(-2*t*t*t+3*t*t)*nodes[k+1][key]+(t*t*t-t*t)*duration*tangent(k+1,key));
0126 }
